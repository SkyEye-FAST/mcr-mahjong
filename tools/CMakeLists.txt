cmake_minimum_required(VERSION 3.20)
project(mcr_oracle LANGUAGES CXX)

# Development-only build. Never included in the Kotlin library or ordinary CI.
set(upstream "${CMAKE_CURRENT_LIST_DIR}/../.reference/upstream")
string(ASCII 239 187 191 bom)
file(READ "${CMAKE_CURRENT_LIST_DIR}/upstream.sha256" manifest)
string(REPLACE "\r\n" "\n" manifest "${manifest}")
string(SHA256 fingerprint "${manifest}")
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/upstream.sha256" checksums)
foreach(entry IN LISTS checksums)
    if(entry MATCHES "^#" OR entry STREQUAL "")
        continue()
    endif()
    string(SUBSTRING "${entry}" 0 64 expected)
    string(SUBSTRING "${entry}" 66 -1 name)
    file(READ "${upstream}/${name}" source)
    string(REGEX REPLACE "^${bom}" "" source "${source}")
    string(REPLACE "\r\n" "\n" source "${source}")
    string(SHA256 actual "${source}")
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Upstream file differs from the revision in NOTICE: ${name}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${upstream}/${name}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/upstream.sha256")

add_executable(oracle oracle.cpp "${upstream}/fan_calculator.cpp"
    "${upstream}/shanten.cpp" "${upstream}/stringify.cpp")
target_include_directories(oracle PRIVATE "${upstream}")
target_compile_features(oracle PRIVATE cxx_std_17)
target_compile_definitions(oracle PRIVATE MCR_UPSTREAM_FINGERPRINT="${fingerprint}")
if(MSVC)
    target_compile_options(oracle PRIVATE /utf-8)
endif()
